function submitPlanForm(e) { e.preventDefault(); const planTexto = document.getElementById('plan-seleccionado').value; // Ej: "Mensualidad - $450" const nombre = document.getElementById('cliente-nombre').value; const telefono = document.getElementById('cliente-telefono').value; const metodoPago = document.getElementById('cliente-metodo-pago').value; // Extraer el nombre del plan y el monto para que el panel de administración los lea perfecto let partesPlan = planTexto.split(' - '); let nombrePlan = partesPlan[0]; let montoPlan = partesPlan[1] || '$0'; // Crear el objeto con la estructura exacta que lee tu panel de administración const nuevaInscripcion = { id: Date.now().toString().slice(-4), cliente: nombre, telefono: telefono, plan: nombrePlan, monto: montoPlan, pago: metodoPago, estatus: 'Pendiente', fecha: new Date().toLocaleDateString() }; // Guardar en la llave que utiliza tu panel de administración (kronox_inscripciones o solicitudes) let inscripciones = JSON.parse(localStorage.getItem('kronox_inscripciones')) || []; inscripciones.push(nuevaInscripcion); localStorage.setItem('kronox_inscripciones', JSON.stringify(inscripciones)); // Notificación flotante personalizada con diseño Kronox (Neón Azul) mostrarNotificacionKronox(`¡Solicitud enviada, ${nombre}! Nos pondremos en contacto vía ${metodoPago}.`); closePlanModal(); e.target.reset(); } // Función para mostrar alerta con el estilo neón de la página function mostrarNotificacionKronox(mensaje) { let notif = document.getElementById('kronox-toast-notification'); if (!notif) { notif = document.createElement('div'); notif.id = 'kronox-toast-notification'; notif.style.cssText = "position:fixed; bottom:30px; right:30px; background:#091122; border:2px solid #00d2ff; color:#fff; padding:16px 24px; border-radius:12px; font-size:14px; font-weight:bold; z-index:99999; box-shadow:0 0 25px rgba(0,210,255,0.5); transition: opacity 0.5s ease; display:flex; align-items:center; gap:12px;"; document.body.appendChild(notif); } notif.innerHTML = ` ${mensaje}`; notif.style.opacity = '1'; setTimeout(() => { notif.style.opacity = '0'; }, 4000); }
